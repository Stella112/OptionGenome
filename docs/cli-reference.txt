  asset_class: "corporate" | "crypto" | "crypto_perp" | "global_equity" | "ipo" | "treasury" | "us_equity" | "us_equity_chain" | "us_index" | "us_option"; // this represents the category to which the asset belongs to
  asset_id: string; // asset ID (For options this represents the option contract ID)
  canceled_at: string; // canceled at
  client_order_id: string; // client unique order ID
  created_at: string; // created at
  expired_at: string; // expired at
  expires_at: string; // expires at
  extended_hours: boolean; // if true, eligible for execution outside regular trading hours
  failed_at: string; // failed at
  filled_at: string; // filled at
  filled_avg_price: string; // filled average price
  filled_qty: string; // filled quantity
  hwm: string; // highest (lowest) market price seen since the trailing stop order was submitted
  id: string; // order ID
  legs: object[]; // when querying non-simple order_class orders in a nested style, an array of Order entities associated with this order
  limit_price: string; // limit price
  notional: string; // ordered notional amount. If entered, qty will be null. Can take up to 9 decimal points
  order_class: "bracket" | "mleg" | "oco" | "oto" | "simple"; // order classes supported by Alpaca vary based on the order's security type
  order_type: string; // deprecated in favour of the field "type"
  position_intent: "buy_to_close" | "buy_to_open" | "sell_to_close" | "sell_to_open"; // represents the desired position strategy
  qty: string; // ordered quantity. If entered, notional will be null. Can take up to 9 decimal points. Required if order class is mleg
  ratio_qty: string; // proportional quantity of this leg in relation to the overall multi-leg order quantity
  replaced_at: string; // replaced at
  replaced_by: string; // order ID that this order was replaced by
  replaces: string; // order ID that this order replaces
  side: "buy" | "sell"; // represents which side this order was on:
  status: "accepted" | "accepted_for_bidding" | "calculated" | "canceled" | "done_for_day" | "expired" | "filled" | "held" | "new" | "partially_filled" | "pending_cancel" | "pending_new" | "pending_replace" | "rejected" | "replaced" | "stopped" | "suspended"; // an order executed through Alpaca can experience several status changes during its lifecycle
  stop_price: string; // stop price
  submitted_at: string; // submitted at
  symbol: string; // asset symbol, required for all order classes except for mleg
  time_in_force: "cls" | "day" | "fok" | "gtc" | "ioc" | "opg"; // time-In-Force values supported by Alpaca vary based on the order's security type
  trail_percent: string; // percent value away from the high water mark for trailing stop orders
  trail_price: string; // dollar value away from the high water mark for trailing stop orders
  type: "limit" | "market" | "stop" | "stop_limit" | "trailing_stop"; // order types supported by Alpaca vary based on the order's security type
  updated_at: string; // updated at
}
